#include <Arduino.h>
#include <Wire.h>
#include <SensirionI2CSdp.h>

SensirionI2CSdp sdp;

void setup() {
  Serial.begin(115200);
  Wire.begin(4, 21);            // SDA=D4, SCL=D21

  char msg[64];
  uint16_t error;

  sdp.begin(Wire, 0x25);        // dirección por defecto del SDP810

  // por si quedó en modo continuo de un reset anterior
  sdp.stopContinuousMeasurement();
  delay(25);

  error = sdp.startContinuousMeasurementWithDiffPressureTCompAndAveraging();
  if (error) {
    errorToString(error, msg, sizeof(msg));
    Serial.print("Error al iniciar: ");
    Serial.println(msg);
  }
  delay(100);  // primera muestra lista
}

void loop() {
  float pressure = 0.0;
  float temperature = 0.0;

  uint16_t error = sdp.readMeasurement(pressure, temperature);
  if (error) {
    char msg[64];
    errorToString(error, msg, sizeof(msg));
    Serial.print("Error lectura: ");
    Serial.println(msg);
  } else {
    Serial.print("P: ");
    Serial.print(pressure, 3);
    Serial.print(" Pa\tT: ");
    Serial.print(temperature, 2);
    Serial.println(" C");
  }
  delay(200);
}
